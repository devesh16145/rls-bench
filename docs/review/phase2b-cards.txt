REQUIREMENT (verbatim)
A team app built with supabase-js. Users create organizations and manage who belongs to them. To create an organization the app calls `supabase.from('orgs').insert({ name }).select()` (so the new row is read back immediately), then inserts the creator's membership with role `owner`. Other writes also use `.select()` to read back the written row.

R1. Any signed-in user can create an organization; its `created_by` is always themselves.
R2. The creator of an organization that has no members yet can add themselves to it as `owner`. Apart from this, nobody can add themselves to an organization.
R3. Members can see the organizations they belong to and every membership row of those organizations. The creator of an organization can also see it before joining it.
R4. Owners and admins can add people with role `member` and remove memberships whose role is `member`. Only owners can give someone the role `admin` or `owner`, change someone's role, or remove admins and owners.
R5. Any member can leave an organization by deleting their own membership, except an organization's only owner.
R6. Nobody can change their own role.
R7. Only owners can rename their organization. Organizations cannot be deleted through the app.
R8. Anonymous visitors cannot see or change anything.
R9. Server-side code using the service role keeps full access.

SOLUTION (verbatim)
create schema private;

grant usage on schema private to authenticated;

create function private.org_role(p_org_id bigint) returns text
language sql stable security definer set search_path = ''
as $$
  select m.role from public.org_members m
  where m.org_id = p_org_id and m.user_id = auth.uid()
$$;

create function private.can_bootstrap(p_org_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.orgs o where o.id = p_org_id and o.created_by = auth.uid())
    and not exists (select 1 from public.org_members m where m.org_id = p_org_id)
$$;

create function private.has_other_owner(p_org_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (
    select 1 from public.org_members m
    where m.org_id = p_org_id and m.role = 'owner' and m.user_id <> auth.uid()
  )
$$;

alter table public.orgs enable row level security;
alter table public.org_members enable row level security;

create policy orgs_select on public.orgs
  for select to authenticated
  using (created_by = auth.uid() or private.org_role(id) is not null);

create policy orgs_insert on public.orgs
  for insert to authenticated
  with check (created_by = auth.uid());

create policy orgs_update on public.orgs
  for update to authenticated
  using (private.org_role(id) = 'owner')
  with check (private.org_role(id) = 'owner');

create policy org_members_select on public.org_members
  for select to authenticated
  using (user_id = auth.uid() or private.org_role(org_id) is not null);

create policy org_members_insert on public.org_members
  for insert to authenticated
  with check (
    (user_id = auth.uid() and role = 'owner' and private.can_bootstrap(org_id))
    or (user_id <> auth.uid() and role = 'member' and private.org_role(org_id) in ('owner', 'admin'))
    or (user_id <> auth.uid() and private.org_role(org_id) = 'owner')
  );

create policy org_members_update on public.org_members
  for update to authenticated
  using (user_id <> auth.uid() and private.org_role(org_id) = 'owner')
  with check (user_id <> auth.uid() and private.org_role(org_id) = 'owner');

create policy org_members_delete on public.org_members
  for delete to authenticated
  using (
    (user_id = auth.uid() and (role <> 'owner' or private.has_other_owner(org_id)))
    or (user_id <> auth.uid() and role = 'member' and private.org_role(org_id) in ('owner', 'admin'))
    or (user_id <> auth.uid() and private.org_role(org_id) = 'owner')
  );

SOLUTION (verbatim)
create schema private;

grant usage on schema private to authenticated;

create function private.is_finance() returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.finance_staff f where f.user_id = auth.uid())
$$;

create function private.invoice_guard() returns trigger
language plpgsql security definer set search_path = ''
as $$
begin
  if auth.role() = 'service_role' then
    return new;
  end if;
  if new.created_by is distinct from old.created_by then
    raise exception 'created_by cannot change';
  end if;
  if old.status <> 'draft'
     and (new.customer is distinct from old.customer or new.amount is distinct from old.amount) then
    raise exception 'invoice % is locked', old.id;
  end if;
  if new.status is distinct from old.status then
    if old.status = 'draft' and new.status = 'sent' and old.created_by = auth.uid() then
      return new;
    end if;
    if old.status = 'sent' and new.status in ('paid', 'void') and private.is_finance() then
      return new;
    end if;
    raise exception 'status change % -> % not allowed', old.status, new.status;
  end if;
  return new;
end
$$;

create trigger invoices_guard
  before update on public.invoices
  for each row execute function private.invoice_guard();

alter table public.finance_staff enable row level security;
alter table public.invoices enable row level security;

create policy finance_staff_select on public.finance_staff
  for select to authenticated
  using (user_id = auth.uid());

create policy invoices_select on public.invoices
  for select to authenticated
  using (created_by = auth.uid() or private.is_finance());

create policy invoices_insert on public.invoices
  for insert to authenticated
  with check (created_by = auth.uid() and status = 'draft');

create policy invoices_update on public.invoices
  for update to authenticated
  using ((created_by = auth.uid() and status = 'draft') or (private.is_finance() and status = 'sent'))
  with check (created_by = auth.uid() or private.is_finance());

create policy invoices_delete on public.invoices
  for delete to authenticated
  using (created_by = auth.uid() and status = 'draft');

A calendar app built with supabase-js. Each event sits on its owner's calendar (`owner_id`). Calendar owners can delegate their calendar to assistants through `delegations`; a delegation may allow editing and may expire at `expires_at` (no value means it never expires). Events are created with `.insert({...}).select()` and edited with `.update({...}).eq('id', id).select()`, so written rows are read back immediately.

R1. Users can create, read, edit and delete events on their own calendar.
R2. An event can never be moved to another calendar: `owner_id` never changes.
R3. A user holding an unexpired delegation can read the grantor's events that are not private. If the delegation allows editing, they can also create, edit and delete the grantor's non-private events, but can never create private events or make an event private.
R4. Private events are never visible to delegates in `events`. Provide a view `public.busy_slots` with exactly the columns `owner_id`, `starts_at` and `ends_at` that lists every event (private or not) on the viewer's own calendar and on the calendars they hold an unexpired delegation for, and nothing else.
R5. Calendar owners can create, change and remove delegations of their own calendar. Delegates can see the delegations given to them. Nobody can create or change delegations of a calendar that isn't theirs, so delegates cannot pass access on.
R6. Anonymous visitors cannot see or change anything, including `busy_slots`.
R7. Server-side code using the service role keeps full access, including to `busy_slots`.

SOLUTION (verbatim)
create schema private;

grant usage on schema private to anon, authenticated;

create function private.has_delegation(p_grantor_id uuid, p_edit boolean) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (
    select 1 from public.delegations d
    where d.grantor_id = p_grantor_id
      and d.delegate_id = auth.uid()
      and (d.expires_at is null or d.expires_at > now())
      and (d.can_edit or not p_edit)
  )
$$;

create function private.event_owner(p_event_id bigint) returns uuid
language sql stable security definer set search_path = ''
as $$
  select e.owner_id from public.events e where e.id = p_event_id
$$;

alter table public.events enable row level security;
alter table public.delegations enable row level security;

create policy events_select on public.events
  for select to authenticated
  using (owner_id = auth.uid() or (not is_private and private.has_delegation(owner_id, false)));

create policy events_insert on public.events
  for insert to authenticated
  with check (owner_id = auth.uid() or (not is_private and private.has_delegation(owner_id, true)));

create policy events_update on public.events
  for update to authenticated
  using (owner_id = auth.uid() or (not is_private and private.has_delegation(owner_id, true)))
  with check (
    owner_id = private.event_owner(id)
    and (owner_id = auth.uid() or (not is_private and private.has_delegation(owner_id, true)))
  );

create policy events_delete on public.events
  for delete to authenticated
  using (owner_id = auth.uid() or (not is_private and private.has_delegation(owner_id, true)));

create policy delegations_select on public.delegations
  for select to authenticated
  using (grantor_id = auth.uid() or delegate_id = auth.uid());

create policy delegations_insert on public.delegations
  for insert to authenticated
  with check (grantor_id = auth.uid());

create policy delegations_update on public.delegations
  for update to authenticated
  using (grantor_id = auth.uid())
  with check (grantor_id = auth.uid());

create policy delegations_delete on public.delegations
  for delete to authenticated
  using (grantor_id = auth.uid());

create view public.busy_slots as
  select e.owner_id, e.starts_at, e.ends_at
  from public.events e
  where e.owner_id = auth.uid()
     or private.has_delegation(e.owner_id, false)
     or auth.role() = 'service_role';
