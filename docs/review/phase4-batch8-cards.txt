SOLUTION (verbatim)
create schema private;

grant usage on schema private to authenticated;

create function private.is_mechanic(p_garage_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.mechanics m where m.garage_id = p_garage_id and m.user_id = auth.uid())
$$;

create function private.job_status(p_job_id bigint) returns text
language sql stable security definer set search_path = ''
as $$
  select j.status from public.jobs j where j.id = p_job_id
$$;

create function private.job_unchanged(p_job_id bigint, p_garage_id bigint, p_customer_id uuid, p_vehicle text, p_quote numeric) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (
    select 1 from public.jobs j
    where j.id = p_job_id and j.garage_id = p_garage_id and j.customer_id = p_customer_id
      and j.vehicle = p_vehicle and j.quote_amount = p_quote
  )
$$;

alter table public.garages enable row level security;
alter table public.mechanics enable row level security;
alter table public.jobs enable row level security;

create policy garages_select on public.garages
  for select to authenticated
  using (true);

create policy jobs_select on public.jobs
  for select to authenticated
  using (customer_id = auth.uid() or private.is_mechanic(garage_id));

create policy jobs_insert on public.jobs
  for insert to authenticated
  with check (private.is_mechanic(garage_id) and status = 'quoted');

create policy jobs_mechanic_update on public.jobs
  for update to authenticated
  using (private.is_mechanic(garage_id))
  with check (
    private.is_mechanic(garage_id)
    and (
      (private.job_status(id) = 'quoted' and status = 'quoted')
      or (
        private.job_status(id) in ('approved', 'in_progress')
        and status in ('in_progress', 'done')
        and private.job_unchanged(id, garage_id, customer_id, vehicle, quote_amount)
      )
    )
  );

create policy jobs_customer_update on public.jobs
  for update to authenticated
  using (customer_id = auth.uid() and status = 'quoted')
  with check (
    customer_id = auth.uid()
    and status = 'approved'
    and private.job_unchanged(id, garage_id, customer_id, vehicle, quote_amount)
  );

create function public.redeem_referral(p_code text) returns bigint
language plpgsql security definer set search_path = ''
as $$
declare
  v_uid uuid := auth.uid();
  v_referrer uuid;
  v_id bigint;
begin
  if v_uid is null then
    raise exception 'not signed in';
  end if;
  select c.user_id into v_referrer from public.referral_codes c where c.code = p_code;
  if not found then
    raise exception 'unknown referral code';
  end if;
  if v_referrer = v_uid then
    raise exception 'cannot use your own referral code';
  end if;
  if exists (select 1 from public.referrals r where r.referee_id = v_uid) then
    raise exception 'you have already used a referral code';
  end if;
  if (select count(*) from public.referrals r where r.referrer_id = v_referrer) >= 5 then
    raise exception 'this referral code has reached its limit';
  end if;
  insert into public.referrals (referrer_id, referee_id) values (v_referrer, v_uid) returning id into v_id;
  insert into public.credits (user_id, amount, reason) values
    (v_uid, 100, 'referred'),
    (v_referrer, 100, 'referral bonus');
  return v_id;
end
$$;

create function public.restore_record(p_record_id bigint) returns void
language plpgsql security definer set search_path = ''
as $$
declare
  v_record public.records%rowtype;
begin
  select * into v_record from public.records where id = p_record_id for update;
  if not found then
    raise exception 'unknown record';
  end if;
  if not exists (
    select 1 from public.org_members m
    where m.org_id = v_record.org_id and m.user_id = auth.uid() and m.role = 'admin'
  ) then
    raise exception 'only admins can restore records';
  end if;
  if v_record.deleted_at is null then
    raise exception 'record is not deleted';
  end if;
  if v_record.deleted_at < now() - interval '30 days' then
    raise exception 'record was deleted more than 30 days ago';
  end if;
  update public.records set deleted_at = null where id = p_record_id;
end
$$;
