create function private.is_reviewer() returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.reviewers r where r.user_id = auth.uid())
$$;

create function private.proposal_speaker(p_proposal_id bigint) returns uuid
language sql stable security definer set search_path = ''
as $$
  select p.speaker_id from public.proposals p where p.id = p_proposal_id
$$;

create function private.proposal_status(p_proposal_id bigint) returns text
language sql stable security definer set search_path = ''
as $$
  select p.status from public.proposals p where p.id = p_proposal_id

create policy reviews_select on public.reviews
  for select to authenticated
  using (

    or (private.proposal_speaker(proposal_id) = auth.uid() and private.proposal_status(proposal_id) <> 'submitted')

create function public.transfer_stock(p_from bigint, p_to bigint, p_sku text, p_qty integer) returns integer
language plpgsql security definer set search_path = ''
as $$
declare
  v_uid uuid := auth.uid();
  v_left integer;
begin
  if v_uid is null or not exists (select 1 from public.warehouse_staff s where s.warehouse_id = p_from and s.user_id = v_uid) then

  end if;

  end if;

  end if;
  update public.stock set qty = qty - p_qty
  where warehouse_id = p_from and sku = p_sku and qty >= p_qty
  returning qty into v_left;
  if not found then
    raise exception 'not enough stock';
  end if;
  insert into public.stock (warehouse_id, sku, qty) values (p_to, p_sku, p_qty)
  on conflict (warehouse_id, sku) do update set qty = public.stock.qty + excluded.qty;
  insert into public.transfers (from_warehouse, to_warehouse, sku, qty, moved_by)
  values (p_from, p_to, p_sku, p_qty, v_uid);
  return v_left;
end

language plpgsql security definer set search_path = ''
as $$
declare
  v_uid uuid := auth.uid();
  v_target_role text;
begin
  if v_uid is null or not exists (
    select 1 from public.org_members m

  end if;

  end if;

end
