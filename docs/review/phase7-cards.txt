SOLUTION (verbatim)
create schema private;

grant usage on schema private to authenticated;

create function private.is_coordinator() returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.coordinators c where c.user_id = auth.uid())
$$;

create function private.shift_open_future(p_shift_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.shifts s where s.id = p_shift_id and s.status = 'open' and s.starts_at > now())
$$;

create function private.shift_has_room(p_shift_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$
  select (select count(*) from public.signups g where g.shift_id = p_shift_id and g.status = 'active') < (select s.capacity from public.shifts s where s.id = p_shift_id)
$$;

create function private.already_signed_up(p_shift_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.signups g where g.shift_id = p_shift_id and g.volunteer_id = auth.uid() and g.status = 'active')
$$;

alter table public.coordinators enable row level security;
alter table public.shifts enable row level security;
alter table public.signups enable row level security;

revoke update on public.signups from anon, authenticated;
grant update (status) on public.signups to authenticated;

create policy coordinators_select on public.coordinators
  for select to authenticated
  using (user_id = auth.uid());

create policy shifts_select on public.shifts
  for select to authenticated
  using (status <> 'cancelled' or private.is_coordinator());

create policy shifts_insert on public.shifts
  for insert to authenticated
  with check (private.is_coordinator() and created_by = auth.uid());

create policy shifts_update on public.shifts
  for update to authenticated
  using (private.is_coordinator())
  with check (private.is_coordinator());

create policy signups_select on public.signups
  for select to authenticated
  using (volunteer_id = auth.uid() or private.is_coordinator());

create policy signups_insert on public.signups
  for insert to authenticated
  with check (
    volunteer_id = auth.uid()
    and status = 'active'
    and private.shift_open_future(shift_id)
    and private.shift_has_room(shift_id)
    and not private.already_signed_up(shift_id)
  );

create policy signups_update on public.signups
  for update to authenticated
  using (volunteer_id = auth.uid() and status = 'active')
  with check (volunteer_id = auth.uid() and status = 'cancelled');

create schema private;

grant usage on schema private to authenticated;

create function private.is_staff() returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.staff s where s.user_id = auth.uid())
$$;

create function private.bike_free(p_bike_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.bikes b where b.id = p_bike_id and b.status = 'in_service') and not exists (select 1 from public.rentals r where r.bike_id = p_bike_id and r.returned_at is null)
$$;

create function private.has_open_rental() returns boolean
language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.rentals r where r.user_id = auth.uid() and r.returned_at is null)
$$;

alter table public.staff enable row level security;
alter table public.bikes enable row level security;
alter table public.rentals enable row level security;

revoke insert, update on public.rentals from anon, authenticated;
grant insert (bike_id, user_id, started_at) on public.rentals to authenticated;
grant update (returned_at) on public.rentals to authenticated;

create policy staff_select on public.staff
  for select to authenticated
  using (user_id = auth.uid());

create policy bikes_select on public.bikes
  for select to authenticated
  using (true);

create policy bikes_update on public.bikes
  for update to authenticated
  using (private.is_staff())
  with check (private.is_staff());

create policy rentals_select on public.rentals
  for select to authenticated
  using (user_id = auth.uid() or private.is_staff());

create policy rentals_insert on public.rentals
  for insert to authenticated
  with check (
    user_id = auth.uid()
    and returned_at is null
    and private.bike_free(bike_id)
    and not private.has_open_rental()
  );

create policy rentals_update on public.rentals
  for update to authenticated
  using (user_id = auth.uid() and returned_at is null)
  with check (user_id = auth.uid() and returned_at is not null);

language sql stable security definer set search_path = ''

language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.submissions s where s.bounty_id = p_bounty_id and s.submitter_id = auth.uid())
$$;

create function private.submission_of_bounty(p_submission_id bigint, p_bounty_id bigint) returns boolean
language sql stable security definer set search_path = ''
as $$

$$;

alter table public.submissions enable row level security;

  for select to authenticated

  for update to authenticated
  using (poster_id = auth.uid() and status = 'open')

    and (
      (status = 'open' and awarded_submission_id is null)

    )
  );

  for select to authenticated

  with check (
    submitter_id = auth.uid()

    and private.bounty_poster(bounty_id) <> auth.uid()
    and not private.already_submitted(bounty_id)
